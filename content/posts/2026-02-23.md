```js
let pairCount = 3; // 현재 열(세로) 세트 개수 (초기 3세트)
let rowCount = 1;  // 현재 행(가로) 개수 (초기 1줄)

const tableBody = document.getElementById('tableBody');
const fullSpanTd = document.getElementById('fullSpanTd');
const animDelay = 400; // CSS transition 시간과 동일하게 설정 (ms)

// 공통 함수: Colspan 업데이트
function updateColspan(changeAmount) {
  const currentColspan = parseInt(fullSpanTd.getAttribute('colspan'), 10);
  fullSpanTd.setAttribute('colspan', currentColspan + changeAmount);
}

// ==========================================
// 1. 행(Row) 추가 / 삭제
// ==========================================
document.getElementById('addRowBtn').addEventListener('click', () => {
  rowCount++;
  const newRow = document.createElement('tr');
  newRow.className = 'data-row';
  const wrappersToAnimate = [];

  // 현재 열(pairCount) 개수만큼 th, td 세트 생성
  for (let i = 1; i <= pairCount; i++) {
    const newTh = document.createElement('th');
    const thWrapper = document.createElement('div');
    thWrapper.className = 'cell-wrapper';
    thWrapper.textContent = `항목 ${rowCount}-${i}`;
    newTh.appendChild(thWrapper);

    const newTd = document.createElement('td');
    const tdWrapper = document.createElement('div');
    tdWrapper.className = 'cell-wrapper';
    tdWrapper.textContent = `데이터 ${rowCount}-${i}`;
    newTd.appendChild(tdWrapper);

    newRow.appendChild(newTh);
    newRow.appendChild(newTd);
    wrappersToAnimate.push(thWrapper, tdWrapper);
  }

  tableBody.appendChild(newRow);

  requestAnimationFrame(() => {
    requestAnimationFrame(() => {
      wrappersToAnimate.forEach(w => w.classList.add('show'));
    });
  });
});

document.getElementById('removeRowBtn').addEventListener('click', () => {
  if (rowCount <= 1) return alert("최소 1개의 행은 남겨야 합니다.");

  const lastRow = tableBody.lastElementChild;
  const wrappers = lastRow.querySelectorAll('.cell-wrapper');

  wrappers.forEach(w => w.classList.remove('show'));

  setTimeout(() => {
    lastRow.remove();
  }, animDelay);

  rowCount--;
});

// ==========================================
// 2. 열(Col) 맨 끝 추가 / 삭제
// ==========================================
document.getElementById('addColBtn').addEventListener('click', () => {
  insertColAtIndex(pairCount + 1); // 맨 끝 인덱스로 추가 함수 호출
});

document.getElementById('removeColBtn').addEventListener('click', () => {
  removeColAtIndex(pairCount); // 맨 끝 인덱스 삭제 함수 호출
});

// ==========================================
// 3. 특정 위치 열(Col) 끼워넣기 / 빼기
// ==========================================
document.getElementById('insertColBtn').addEventListener('click', () => {
  const targetNumber = parseInt(document.getElementById('colIndexInput').value, 10);
  insertColAtIndex(targetNumber);
});

document.getElementById('removeSpecificColBtn').addEventListener('click', () => {
  const targetNumber = parseInt(document.getElementById('colIndexInput').value, 10);
  removeColAtIndex(targetNumber);
});

// --- 핵심 로직: 특정 위치에 열 추가 함수 ---
function insertColAtIndex(targetNumber) {
  if (targetNumber < 1 || targetNumber > pairCount + 1) {
    return alert("현재 끼워넣을 수 없는 위치입니다.");
  }

  pairCount++;
  const dataRows = document.querySelectorAll('.data-row');
  const wrappersToAnimate = [];

  dataRows.forEach((row, rowIndex) => {
    const newTh = document.createElement('th');
    const thWrapper = document.createElement('div');
    thWrapper.className = 'cell-wrapper';
    thWrapper.textContent = `새 항목 (행${rowIndex+1})`;
    newTh.appendChild(thWrapper);

    const newTd = document.createElement('td');
    const tdWrapper = document.createElement('div');
    tdWrapper.className = 'cell-wrapper';
    tdWrapper.textContent = `새 데이터`;
    newTd.appendChild(tdWrapper);

    // 타겟 요소 찾기
    const targetIndex = (targetNumber - 1) * 2;
    const targetElement = row.children[targetIndex];

    if (targetElement) {
      row.insertBefore(newTh, targetElement);
      row.insertBefore(newTd, targetElement);
    } else {
      row.appendChild(newTh);
      row.appendChild(newTd);
    }

    wrappersToAnimate.push(thWrapper, tdWrapper);
  });

  updateColspan(2);

  requestAnimationFrame(() => {
    requestAnimationFrame(() => {
      wrappersToAnimate.forEach(w => w.classList.add('show'));
    });
  });
}

// --- 핵심 로직: 특정 위치 열 삭제 함수 ---
function removeColAtIndex(targetNumber) {
  if (pairCount <= 1) return alert("최소 1세트의 열은 남겨야 합니다.");
  if (targetNumber < 1 || targetNumber > pairCount) return alert("삭제할 수 없는 위치입니다.");

  const dataRows = document.querySelectorAll('.data-row');

  dataRows.forEach(row => {
    // 지울 대상 찾기 (th와 td 세트)
    const targetThIndex = (targetNumber - 1) * 2;
    const targetTh = row.children[targetThIndex];
    const targetTd = row.children[targetThIndex + 1];

    if (targetTh && targetTd) {
      // 1. 애니메이션 트리거 (스르륵 사라짐)
      targetTh.querySelector('.cell-wrapper').classList.remove('show');
      targetTd.querySelector('.cell-wrapper').classList.remove('show');

      // 2. 시간차 DOM 삭제
      setTimeout(() => {
        targetTh.remove();
        targetTd.remove();
      }, animDelay);
    }
  });

  // colspan은 DOM이 다 사라진 후(animDelay)가 아니라, 애니메이션 시작과 동시에 줄여야 레이아웃이 안 깨집니다.
  setTimeout(() => updateColspan(-2), animDelay);
  pairCount--;
}
```
